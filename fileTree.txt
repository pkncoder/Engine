.
├── assets/                                 # Scenes, models, textures, and HDRI maps
│   ├── materials/                             # Material files (.mat)
│   ├── models/                                # Model files (.obj)
│   └── textures/                              # Texture files (seperated into directories matching the model file name)
├── debugShaders/                           # The last compiled fully concatenated shaders from the #includes (in text)
│   ├── last_COMPUTE_gbuffer.comp.glsl          # Final compute shader
│   ├── last_COMPUTE_invert.comp.glsl           # Final compute shader
│   ├── last_COMPUTE_pathTracer.comp.glsl       # Final compute shader
│   ├── last_FRAGMENT_raster.frag.glsl          # Final fragment shader
│   └── last_VERTEX_raster.vert.glsl            # Final vertex shader
├── build/                                  # Generated build files (gitignored)
├── shaders/                                # GPU-Code
│   ├── compute/                                # Ray-Based Rendering shaders (.comp)
│   │   └── invert.comp                             # Testing post-processing effect for shader passes
│   ├── include/                                # Shared "library" files (.glsl)
│   │   ├── modelBases/                             # Base math for each shading model + a base function to get the final color
│   │   │   ├── blinnPhongBase.glsl                     # Blinn phong base model
│   │   │   └── cookTorranceBDRFBase.h                  # Cook-Torrance BDRF base model
│   │   ├── utils/                                  # Utility functions
│   │   │   ├── SRGB.glsl                               # SRGB translation functions
│   │   │   ├── random.glsl                             # Random lib utility
│   │   │   └── toneMapping.h                           # Tone mapping functions
│   │   ├── common.glsl                             # Common definitions, usually for things like PI & EPSILON
│   │   ├── sharedStructures.glsl                   # Structures shared throughout every renderer
│   │   └── sharedUniforms.glsl                     # Uniforms shared throughout every renderer
│   ├── pathTracing/                            # Path tracer renderer
│   │   ├── main/                                   # Shader files that the renderers connect with
│   │   │   ├── gbuffer.comp                            # First hit data saving
│   │   │   └── pathTracer.comp                         # Main render file for the path tracer
│   │   ├── models/                                 # Translation files from path tracing data to base function models
│   │   │   ├── blinnPhong.glsl                         # Main entry Blinn-Phong function to translate path tracing data to the base
│   │   │   └── cookTorranceBDRF.glsl                   # Main entry BDRF function to translate path tracing data to the base
│   │   ├── buffers.glsl                            # Data buffers
│   │   ├── intersections.glsl                      # Ray-Intersection functions
│   │   ├── structures.glsl                         # Structure definitions
│   │   ├── to_be_uniformed_valeus.glsl             # Temporary file for definitions that will eventually be uniforms
│   │   └── uniforms.glsl                           # Uniforms
│   └── rasterizing/                           # Rasterizer renderer
│       ├── main/                                   # Shader files that the renderers connect with
│       │   ├── raster.frag                             # Vertex shader for the rasterizer
│       │   └── raster.vert                             # Fragment shader for the rasterizer
│       ├── models/                                 # Translation files from rasterization data to base function models
│       │   ├── blinnPhong.glsl                         # Main entry Blinn-Phong function to translate rasterization data to the base
│       │   └── cookTorranceBDRF.glsl                   # Main entry BDRF function to translate rasterization data to the base
│       └── uniforms.glsl                           # Uniforms
├── src/                                    # CPU-Code
│   ├── core/                                   # Engine internals
│   │   ├── Application.cpp/h                       # Main loop & layer management
│   │   └── Window.cpp/h                            # GLFW abstraction
│   ├── renderer/                               # Rendering Logic & GPU Data handling
│   │   ├── BufferManager.cpp/h                     # SSBO/VBO/VAO handling
│   │   ├── GPUStructs.h                            # Struct declarations for anything that deals with sending data to the GPU. Similar to CPUStructs.h
│   │   ├── IRenderer.h                             # Base interface for a renderer
│   │   ├── PathTracer.cpp/h                        # GPU Path Tracing
│   │   ├── Rasterizer.cpp/h                        # GPU Rasterizer
│   │   ├── RendererManager.cpp/h                   # Renderer management
│   │   └── Shader.cpp/h                            # OpenGL shader and program compilation & usage
│   ├── resources/                              # Resource loaders & managers
│   │   ├── AssetManager.cpp/h                      # The "Central Hub" / Cache logic
│   │   ├── CPUStructs.h                            # Struct declarations for the AssetManager & co. (anything that deals with CPU Data)
│   │   ├── MaterialLoader.cpp/h                    # Material parsing (tinyobjloader)
│   │   ├── ModelLoader.cpp/h                       # Geometry parsing (tinyobjloader)
│   │   └── TextureLoader.cpp/h                     # Image parsing (stb_image)
│   ├── scene/                                  # Scene and entity logic
│   │   ├── Components/                             # Component Logic
│   │   │   ├── MaterialComponent.h                     # Material information
│   │   │   ├── MeshComponent.h                         # Mesh component glad IDs
│   │   │   └── TransformComponent.h                    # Position, Rotation, Scale information
│   │   ├── Campera.cpp/h                           # Camera logic
│   │   ├── ECS.h                                   # Entity ID typdefs & component registering
│   │   ├── Entity.h                                # Entity wrapper for ease of use
│   │   ├── EntitySpawner.h                         # Helper header file for spawning specific entity types (ex. renderable)
│   │   └── Scene.cpp/h                             # The "Registry" (manages the list of entities)
│   ├── services/                               # Static classes that provide services
│   │   ├── Input.cpp/h                             # Input checking and deltaMouse polling
│   │   ├── Logger.cpp/h                            # Engine-wide static logger for anything required
│   │   └── Timer.ccp/h                             # FPS, profiling, and anything time-based service
│   ├── Constants.h                             # Constant values
│   ├── Defaults.h                              # Default values
│   └── main.cpp                                # Entry point
├── CMakeLists.txt                          # Main build script
└── vcpkg.json                              # Package manager
